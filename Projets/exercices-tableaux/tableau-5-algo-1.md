// Cet algorithme cree un tableau d'entiers nombre de longueur 4. Puis on affiche les entiers au carre //
// Voici la version simplifiee en 1 boucle au lieu de 2 //
VARIABLES
  TABLEAU D'ENTIERS nombres[4]
  ENTIER i
DEBUT
  POUR i allant de 0 à 3
    nombres[i] <-- i * i
    ECRIRE nombres[i]
  FIN POUR
FIN
